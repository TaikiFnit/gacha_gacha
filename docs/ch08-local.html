8.1 myserver.py をパッケージに分割

myserver.py は今や数百行になっているはず。 これを役割ごとに分割します。 最終的な構成は以下のようになります: 

$HOME\gacha_workspace\
├── server\
│   ├── __init__.py        ← 空ファイル (これがあると "パッケージ" 扱いになる)
│   ├── app.py             ← HTTP サーバ + ルート定義
│   ├── db.py              ← psycopg 接続ヘルパ
│   ├── auth.py            ← パスワードハッシュ + セッション
│   └── gacha.py           ← 重み付き乱択
├── web\                   ← Ch.4 でコピー済み
├── db\                    ← Ch.2 で作成
├── docker-compose.yml     ← Ch.2 で作成
└── tests\                 ← この章で追加

分割の方針

新ファイル移すもの
server/db.py DB_PARAMS, get_conn, healthcheck

myserver.py の中身を、 役割ごとに以下の 4 ファイルへコピペで切り出します。 ロジックは何も変えません(同じ関数を別ファイルに動かすだけ)。 変わるのは server/app.py の import 文だけ — fetch_pool や hash_password が他のファイルに移ったので、 gacha.fetch_pool / auth.hash_password のように モジュール名経由で呼び出すように書き換えます。

"""DB 接続ヘルパ (psycopg)。"""
from contextlib import contextmanager

import psycopg
from psycopg.rows import dict_row

DB_PARAMS = dict(
    host="localhost", port=5432, dbname="gacha",
    user="gacha", password="gacha",
)
# 注: 8.2 で .env から読むように変更します


@contextmanager
def get_conn():
    """1 リクエストぶんの DB 接続を貸す。 with ブロック抜けで commit / rollback。"""
    conn = psycopg.connect(**DB_PARAMS, row_factory=dict_row, autocommit=False)
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def healthcheck():
    """DB に届くか確認する。 サーバ起動時の早期失敗用。"""
    try:
        with get_conn() as conn, conn.cursor() as cur:
            cur.execute("SELECT 1")
            cur.fetchone()
        return True
    except Exception as e:
        print(f"[db] healthcheck failed: {e}")
        return False

📝 server/auth.py — パスワード + セッション

myserver.py の hash_password / verify_password + セッション系を切り出し:

"""認証 (パスワードハッシュ + セッショントークン)。"""
import base64
import hashlib
import hmac
import os
import secrets
from datetime import datetime, timedelta, timezone

_PBKDF2_ALGO = "sha256"
_PBKDF2_ITER = 200_000
_PBKDF2_SALT_BYTES = 16
_PBKDF2_DKLEN = 32
SESSION_TTL = timedelta(days=7)


def hash_password(password):
    salt = os.urandom(_PBKDF2_SALT_BYTES)
    dk = hashlib.pbkdf2_hmac(_PBKDF2_ALGO, password.encode("utf-8"),
                             salt, _PBKDF2_ITER, dklen=_PBKDF2_DKLEN)
    return "$".join([
        f"pbkdf2_{_PBKDF2_ALGO}",
        str(_PBKDF2_ITER),
        base64.b64encode(salt).decode(),
        base64.b64encode(dk).decode(),
    ])


def verify_password(password, stored):
    try:
        algo, iter_s, salt_b64, hash_b64 = stored.split("$")
    except ValueError:
        return False
    if not algo.startswith("pbkdf2_"):
        return False
    digest_name = algo.split("_", 1)[1]
    iters = int(iter_s)
    salt = base64.b64decode(salt_b64)
    expected = base64.b64decode(hash_b64)
    actual = hashlib.pbkdf2_hmac(digest_name, password.encode("utf-8"),
                                 salt, iters, dklen=len(expected))
    return hmac.compare_digest(actual, expected)


def new_session_token():
    return secrets.token_urlsafe(32)


def create_session(conn, user_id):
    token = new_session_token()
    expires = datetime.now(timezone.utc) + SESSION_TTL
    with conn.cursor() as cur:
        cur.execute(
            "INSERT INTO sessions (token, user_id, expires_at) VALUES (%s, %s, %s)",
            (token, user_id, expires),
        )
    return token, expires


def lookup_session(conn, token):
    if not token:
        return None
    with conn.cursor() as cur:
        cur.execute("""
            SELECT u.id, u.name, u.display_name, u.coins, s.expires_at
              FROM sessions s
              JOIN users    u ON u.id = s.user_id
             WHERE s.token = %s
        """, (token,))
        row = cur.fetchone()
    if row is None:
        return None
    if row["expires_at"] <= datetime.now(timezone.utc):
        return None
    return {"id": row["id"], "name": row["name"],
            "display_name": row["display_name"], "coins": row["coins"]}


def delete_session(conn, token):
    if not token:
        return
    with conn.cursor() as cur:
        cur.execute("DELETE FROM sessions WHERE token = %s", (token,))

📝 server/gacha.py — ガチャ抽選

myserver.py の GachaItem dataclass + fetch_pool + draw を切り出し:

"""ガチャ抽選ロジック (重み付き乱択)。"""
import random
from dataclasses import dataclass


@dataclass
class GachaItem:
    character_id: int
    name: str
    rarity: int
    emoji: str
    weight: int


def fetch_pool(conn, gacha_id):
    with conn.cursor() as cur:
        cur.execute("""
            SELECT c.id AS character_id, c.name, c.rarity, c.emoji, gi.weight
              FROM gacha_items gi
              JOIN characters c ON c.id = gi.character_id
             WHERE gi.gacha_id = %s
             ORDER BY c.rarity, c.id
        """, (gacha_id,))
        rows = cur.fetchall()
    return [GachaItem(**r) for r in rows]


def draw(pool, rng=None):
    if not pool:
        raise ValueError("ガチャに排出設定がありません")
    rng = rng or random.SystemRandom()
    weights = [item.weight for item in pool]
    return rng.choices(pool, weights=weights, k=1)[0]

"""http.server で書かれたガチャ API サーバ。"""
from __future__ import annotations

import json
import os
import sys
import traceback
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import urlparse

import psycopg

# ↓ 切り出した 3 ファイルから取り込む(相対 import)
from . import auth, gacha
from .db import get_conn, healthcheck

APP_PORT = int(os.getenv("APP_PORT", "8000"))


# ---- 例外 / ルーター --------------------------------------------------
class AppError(Exception):
    def __init__(self, status, message):
        super().__init__(message)
        self.status = status
        self.message = message

def bad_request(msg): return AppError(400, msg)
def unauthorized(msg="ログインが必要です"): return AppError(401, msg)
def not_found(msg="見つかりません"): return AppError(404, msg)

ROUTES = {}
def route(method, path):
    def deco(fn):
        ROUTES[(method.upper(), path)] = fn
        return fn
    return deco


# ---- 共通ヘルパ -------------------------------------------------------
def _user_payload(u):
    return {"id": u["id"], "name": u["name"],
            "display_name": u["display_name"], "coins": u["coins"]}


# ===== ハンドラ ========================================================
@route("POST", "/api/register")
def register(h):
    body = h.read_json()
    name = (body.get("name") or "").strip()
    password = body.get("password") or ""
    display_name = (body.get("display_name") or "").strip() or name
    if len(name) < 3 or len(name) > 32:
        raise bad_request("name は 3〜32 文字")
    if len(password) < 6:
        raise bad_request("password は 6 文字以上")

    pass_hash = auth.hash_password(password)             # ← auth モジュール経由
    try:
        with get_conn() as conn, conn.cursor() as cur:
            cur.execute("SELECT 1 FROM users WHERE name = %s", (name,))
            if cur.fetchone():
                raise bad_request("その name は既に使われています")
            cur.execute(
                "INSERT INTO users (name, pass_hash, display_name) "
                "VALUES (%s, %s, %s) "
                "RETURNING id, name, display_name, coins",
                (name, pass_hash, display_name),
            )
            user = cur.fetchone()
            token, _ = auth.create_session(conn, user["id"])
    except psycopg.errors.UniqueViolation:
        # 競合: 並行リクエストで同名登録された場合の最後の防波堤
        raise bad_request("その name は既に使われています")
    h.send_json(200, {"user": _user_payload(user), "token": token})


@route("POST", "/api/login")
def login(h):
    body = h.read_json()
    name = (body.get("name") or "").strip()
    password = body.get("password") or ""
    with get_conn() as conn, conn.cursor() as cur:
        cur.execute(
            "SELECT id, name, pass_hash, display_name, coins "
            "FROM users WHERE name = %s",
            (name,),
        )
        user = cur.fetchone()
        if user is None or not auth.verify_password(password, user["pass_hash"]):
            raise AppError(401, "name か password が違います")
        token, _ = auth.create_session(conn, user["id"])
    h.send_json(200, {"user": _user_payload(user), "token": token})


@route("POST", "/api/logout")
def logout(h):
    token = h.get_bearer_token()
    if token:
        with get_conn() as conn:
            auth.delete_session(conn, token)
    h.send_json(200, {"ok": True})


@route("GET", "/api/me")
def me(h):

@route("POST", "/api/gacha/pull")
def gacha_pull(h):
    body = h.read_json()
    gacha_id = body.get("gacha_id")
    if not isinstance(gacha_id, int):
        raise bad_request("gacha_id は整数で指定してください")

    with get_conn() as conn:
        user = h.require_user(conn)

        with conn.cursor() as cur:
            cur.execute(
                "SELECT id, name, price FROM gachas WHERE id = %s FOR SHARE",
                (gacha_id,),
            )
            g = cur.fetchone()
            if g is None:
                raise not_found("そのガチャは存在しません")

            cur.execute(
                "SELECT coins FROM users WHERE id = %s FOR UPDATE",
                (user["id"],),
            )
            coins = cur.fetchone()["coins"]
            if coins < g["price"]:
                raise bad_request(
                    f"コインが足りません (必要 {g['price']} / 所持 {coins})"
                )

        pool = gacha.fetch_pool(conn, gacha_id)           # ← gacha モジュール経由
        if not pool:
            raise bad_request("そのガチャには排出設定がありません")
        won = gacha.draw(pool)

        with conn.cursor() as cur:
            cur.execute(
                "UPDATE users SET coins = coins - %s WHERE id = %s "
                "RETURNING coins",
                (g["price"], user["id"]),
            )
            new_coins = cur.fetchone()["coins"]
            cur.execute(
                "INSERT INTO user_characters (user_id, character_id) "
                "VALUES (%s, %s) RETURNING obtained_at",
                (user["id"], won.character_id),
            )
            obtained_at = cur.fetchone()["obtained_at"]

    h.send_json(200, {
        "gacha":     {"id": g["id"], "name": g["name"], "price": g["price"]},
        "character": {"id": won.character_id, "name": won.name,
                      "rarity": won.rarity, "emoji": won.emoji},
        "obtained_at": obtained_at,
        "coins": new_coins,
    })

def box(h):

# ===== HTTP ハンドラ + CORS + ディスパッチ ============================
class GachaHandler(BaseHTTPRequestHandler):
    def log_message(self, fmt, *args):
        sys.stderr.write(f"[{self.command}] {self.path} -> {fmt % args}\n")

    def _cors_headers(self):
        self.send_header("Access-Control-Allow-Origin",  "*")
        self.send_header("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
        self.send_header("Access-Control-Allow-Headers", "Content-Type, Authorization")

    def send_json(self, status, payload):
        body = json.dumps(payload, ensure_ascii=False, default=str).encode("utf-8")
        self.send_response(status)
        self._cors_headers()
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def read_json(self):
        n = int(self.headers.get("Content-Length", "0"))
        if n == 0: return {}
        raw = self.rfile.read(n)
        try:
            data = json.loads(raw.decode("utf-8"))
        except json.JSONDecodeError as e:
            raise bad_request(f"JSON が読めません: {e}")
        if not isinstance(data, dict):
            raise bad_request("JSON の最上位はオブジェクトにしてください")
        return data

    def get_bearer_token(self):
        h = self.headers.get("Authorization", "")
        if not h.lower().startswith("bearer "):
            return ""
        return h.split(" ", 1)[1].strip()

    def require_user(self, conn):
        token = self.get_bearer_token()
        user = auth.lookup_session(conn, token)
        if user is None:
            raise unauthorized()
        return user

    def do_OPTIONS(self):
        self.send_response(204)
        self._cors_headers()
        self.send_header("Content-Length", "0")
        self.end_headers()

    def do_GET(self):  self._dispatch("GET")
    def do_POST(self): self._dispatch("POST")

    def _dispatch(self, method):
        path = urlparse(self.path).path
        handler = ROUTES.get((method, path))
        if handler is None:
            self.send_json(404, {"error": "Not Found", "path": path})
            return
        try:
            handler(self)
        except AppError as e:
            self.send_json(e.status, {"error": e.message})
        except Exception as e:
            traceback.print_exc()
            self.send_json(500, {"error": "internal server error", "detail": str(e)})


def main():
    if not healthcheck():
        print("PostgreSQL に接続できません。 docker compose up -d を先に。")
        return 1
    addr = ("0.0.0.0", APP_PORT)
    server = ThreadingHTTPServer(addr, GachaHandler)
    print(f"[gacha_gacha] listening on http://localhost:{APP_PORT}")
    print("  ルート一覧:")
    for (m, p) in sorted(ROUTES):
        print(f"    {m:5s} {p}")
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nbye")
    finally:
        server.server_close()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

📚 コラム: Python のパッケージ (= フォルダに __init__.py)

Python では「ディレクトリに __init__.py という空ファイルを置くと、 そのディレクトリが パッケージとして扱われる」 という伝統があります (3.3 以降は名前空間パッケージという仕組みもあって __init__.py が必須ではないですが、 明示的に置く方が混乱しません)。

パッケージ内のモジュール同士は 相対 importで参照できます: 

# server/app.py から server/db.py の get_conn を呼ぶ
from .db import get_conn

# server/app.py から server/auth.py 全体を使う
from . import auth
auth.hash_password("...")

パッケージ全体を起動するときは python -m server.app のように -m を付けます。 これで Python が server を「パッケージとして」 扱い、 相対 import が動きます。

起動コマンドが変わる

分割前: python myserver.py
分割後: python -m server.app

▶️ 動作確認

cd $HOME\gacha_workspace
python -m server.app

Ch.7 までと同じ起動メッセージ + ルート一覧が出ればパッケージ化は成功。 Bruno で一通りの リクエストを打って、 機能不変であることを確認します。

⚠️ よくある分割ミス
  • from db import get_conn (相対なし) → エラー。 from .db import get_conn
  • __init__.py を作り忘れる → 「No module named 'server'」
  • 循環 import (例: db.py が auth を import、 auth が db を import) → ImportError。 役割を見直してどちらかに寄せる

8.2 設定を .env に外出し

DB 接続情報をソースコードに直書きしているのを .env ファイルに移します。 これで「本番ではパスワードを変える」 「テストでは別 DB を使う」 などの場面に対応できます。

📝 server/db.py — 環境変数を読むように書き換え

§8.1 で書いた server/db.py を、 以下の内容で丸ごと置き換えます。 DB_PARAMS の直書きが _conn_params() の os.getenv(...) に変わるだけで、 get_conn / healthcheck の役割は同じ。 

"""DB 接続ヘルパ (psycopg)。 環境変数で接続情報を上書きできる。"""
import os
from contextlib import contextmanager

import psycopg
from psycopg.rows import dict_row


def _conn_params() -> dict:
    """環境変数から接続パラメータを組み立てる。 未設定の項目は既定値を使う。"""
    return dict(
        host=os.getenv("PGHOST", "localhost"),
        port=os.getenv("PGPORT", "5432"),
        dbname=os.getenv("PGDATABASE", "gacha"),
        user=os.getenv("PGUSER", "gacha"),
        password=os.getenv("PGPASSWORD", "gacha"),
    )


@contextmanager
def get_conn():
    conn = psycopg.connect(**_conn_params(), row_factory=dict_row, autocommit=False)
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def healthcheck():
    """DB に届くか確認する。 サーバ起動時の早期失敗用。"""
    try:
        with get_conn() as conn, conn.cursor() as cur:
            cur.execute("SELECT 1")
            cur.fetchone()
        return True
    except Exception as e:
        print(f"[db] healthcheck failed: {e}")
        return False
⚠️ healthcheck も忘れずに残す §8.1 で healthcheck も db.py に書いていたはず。 ここで上書きするときに 消し忘れないこと。 server/app.py の冒頭が from .db import get_conn, healthcheck で healthcheck を import するため、 消すと ImportError: cannot import name 'healthcheck' でサーバが起動しない。

.env はgit にコミットしないのが鉄則 (パスワードを含むため)。 リポジトリには .env.example という雛形だけ置いて、 .gitignore に .env を追加するのが定番のパターンです。

8.3 ユニットテスト (DB 不要)

DB を立てなくても動くテストから書きます。 パスワードハッシュの roundtrip と ガチャ抽選の分布。 純粋関数なのでテストしやすい題材です。

📝 tests/test_unit.py 
"""DB-free unit tests for server.auth and server.gacha."""
import pathlib, random, sys
ROOT = pathlib.Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))

from server import auth, gacha

_failed = False

def expect(label, cond, detail=""):
    global _failed
    mark = "PASS" if cond else "FAIL"
    if not cond: _failed = True
    print(f"  {mark}  {label}  {detail if not cond else ''}")

# ---- auth: hash + verify roundtrip --------------------------------------
print("\n[auth] hash + verify roundtrip")
pw = "correct horse battery staple"
h = auth.hash_password(pw)
expect("verify with correct password", auth.verify_password(pw, h))
expect("verify with wrong password rejects", not auth.verify_password("wrong", h))
expect("hash starts with pbkdf2_sha256$", h.startswith("pbkdf2_sha256$"))

# ---- auth: salt makes hashes differ -------------------------------------
print("\n[auth] same password -> different hashes")
h1 = auth.hash_password("abc12345")
h2 = auth.hash_password("abc12345")
expect("two stored hashes differ", h1 != h2)

# ---- gacha: 200,000 回引いて weights に近づくか --------------------------
print("\n[gacha] empirical distribution converges")
pool = [
    gacha.GachaItem(1, "A", 1, "🟢", 50),
    gacha.GachaItem(2, "B", 2, "🦇", 30),
    gacha.GachaItem(3, "C", 3, "🛡️", 15),
    gacha.GachaItem(4, "D", 5, "🐉",  5),
]
total_w = sum(p.weight for p in pool)
rng = random.Random(42)
n = 200_000
counts = {p.character_id: 0 for p in pool}
for _ in range(n):
    counts[gacha.draw(pool, rng=rng).character_id] += 1
for p in pool:
    expected = p.weight / total_w
    actual = counts[p.character_id] / n
    expect(f"char {p.character_id} {actual:.4f} vs {expected:.4f}",
           abs(actual - expected) < 0.01)

print("\n========== summary ==========")
print("  status:", "FAILED" if _failed else "PASSED")
sys.exit(1 if _failed else 0)
▶️ 実行 
python tests\test_unit.py
✅ 期待される出力 
[auth] hash + verify roundtrip
  PASS  verify with correct password
  ...
[gacha] empirical distribution converges
  PASS  char 1 0.4990 vs 0.5000
  ...
========== summary ==========
  status: PASSED

8.4 smoke テスト (本物の Postgres + サーバを叩く)

本物の Postgres + 本物のサーバプロセスに対して、 主要 API を順に叩いて応答を確認する シェルスクリプトを置きます。 リリース前にこれを通せば「最低限の体裁は OK」 と言えます。

📝 smoke_curl.sh は bash 用

本リポジトリには tests/smoke_curl.sh が用意されています。 これは bash スクリプトなので、 Windows では Git Bash もしくは WSLから実行してください。 PowerShell からそのまま bash tests/smoke_curl.sh として呼び出せます (Git Bash がインストールされていれば)。

必要な前提:

  • docker compose up -d で Postgres が起動済み
  • python -m server.app で API サーバが起動済み
  • jq コマンドがインストール済み (Git Bash には付属、 winget でも入る: winget install jqlang.jq)
▶️ 実行 
# Git Bash もしくは PowerShell から
bash tests/smoke_curl.sh

register → me → gacha list → pull × 3 → box → logout → me (401) の一連の流れが 期待通りに走り、 全部のステータスとレスポンスが見られます。

8.5 GitHub Actions で自動実行

毎回手で叩くのは面倒なので、 push / PR のたびに上のテストを CI で走らせます。 リポジトリには .github/workflows/test.yml が用意されています。

📝 .github/workflows/test.yml 
name: tests

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  unit-and-e2e:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.11" }
      - run: pip install "psycopg[binary]" || true
      - run: python tests/test_unit.py
      - run: python tests/test_e2e_sqlite.py    # 開発者向け: SQLite で全 API を回す

tests/test_e2e_sqlite.py は開発者向けの高速 E2E テストで、 Postgres を立てずに SQLite で server/app.py 全体を回します。 受講者本人は 触らなくて構いませんが、 リポジトリの開発者・メンテ担当向けに置いてあります。
